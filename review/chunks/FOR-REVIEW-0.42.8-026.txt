// Independent JavaScript calculation; the native encoder computes its own checksum.
function crc32(bytes) {
  let crc = 0xffffffff;
  for (const byte of bytes) {
    crc ^= byte;
    for (let bit = 0; bit < 8; bit++) crc = (crc >>> 1) ^ ((crc & 1) ? 0xedb88320 : 0);
  }
  return (crc ^ 0xffffffff) >>> 0;
}

function checksum(frame) {
  if (frame.length < 4) return { valid: false, stored: null, computed: null };
  const stored = frame.readUInt32LE(frame.length - 4);
  const computed = crc32(frame.subarray(0, frame.length - 4));
  return { valid: stored === computed, stored, computed };
}

function selectInputs(sourceCases) {
  if (!Array.isArray(sourceCases)) throw new TypeError('sourceCases must be the offline native scenario cases');
  return [
    ['native-race-capture', 'synthetic-race-envelope', 21, 'race'],
    ['native-freeroam-capture', 'synthetic-free-roam-envelope', 22, 'freeroam'],
  ].map(([id, channel, type, label]) => {
    const candidates = sourceCases.filter(item => item.id === id);
    if (candidates.length !== 1) throw new Error(`Missing or duplicate source native case: ${id}`);
    const source = candidates[0];

      throw new Error(`Source native provenance mismatch: ${id}`);
    }
    const captures = source.captures?.filter(item => item.stage === 'pre-encoding' && item.channel === channel);
    if (captures?.length !== 1) throw new Error(`Missing unique pre-encoding source capture: ${id}`);

}

function validateNativeModuleImports(module, id) {

===== offline-report-lab/README.txt =====
OFFLINE REPORT LAB

Run from this folder with Node.js (verified with v26.13.0):

  node lab.cjs

To also execute the original isolated encoder on the captured synthetic reports:

  node lab.cjs --encode

No package installation is needed. Open the index.html path printed by the command.
Each run creates a new directory under runs, with report.json and captures/*.bin.
The HTML and JSON label pre-encoding inputs and encoded outputs separately.
To choose a new or empty output directory:

  node lab.cjs --out runs/my-inspection --json

Run verification:

  node --test --test-concurrency=1 test/*.test.cjs

WHAT EXECUTES

The native cases instantiate eight small isolated helper bodies copied from the
reviewed 60946132 module. Frame observations, the selected record descriptor,
context pointers and session data are controlled fixtures. The complete game
does not load or run. Original native envelope builders call a local recorder
at their encoder-input boundary. This boundary returns zero, so the downstream
sender never runs. Those captures are native input structures before encoding.

With --encode, two captured synthetic inputs are separately replayed through
the original isolated f22860 encoder, with its seven actual leaf helpers linked. The
encoder has no transport implementation. Its returned bytes are saved locally;
an independent checksum check and short-buffer cases inspect the result and
capacity handling. This executes selected serialization paths, not every
message schema, the full client or an actual online report exchange.

The page-check cases use pinned source in an isolated VM. They compare clean
APIs with the actual extension startup wrapper. A separate minimal-module case
demonstrates explicit local loading without replacing browser APIs. That case
does not establish compatibility of a full patched game.

NETWORK BOUNDARY

The CLI blocks common networking, subprocess and worker APIs in its own Node
process. Native fixtures have exact import allowlists; the page-check VM has
no network or process capability. The local HTML viewer has no scripts and a
Content Security Policy that blocks connections and external assets. No HTTP
server is needed. This is not an OS firewall or a general untrusted-code sandbox.

Fixtures and provenance are bundled and hash-pinned. No official-site requests,
accounts, browser sessions, external services or installed-extension changes
are part of a run. Historical reports and private server decisions are absent.
A clean controlled case does not prove multiplayer safety.

CONTINUING THE OFFLINE WORK

lib/local-loader.cjs is the shared entry point. It accepts local bytes and a
complete expected SHA-256, checks the source, applies explicitly supplied patch
steps, checks every step's expected input/output hash, then compiles directly.
It does not fetch files, intercept browser APIs, supply imports, instantiate a
game or start an entry point. Each caller must provide its own bounded imports
and decide whether to instantiate its verified isolated module.

To adapt an existing pure patcher, provide a transform record:

  {
    name: 'Reviewed offline patch',
    inputSha256: VERIFIED_INPUT_HASH,
    outputSha256: VERIFIED_OUTPUT_HASH,
    apply: bytes => existingPatcher(bytes)
  }

The current native fixtures already use the shared loader. Future work can add

#!/usr/bin/env node
'use strict';

const fs = require('node:fs');
const path = require('node:path');
const crypto = require('node:crypto');
const { installNetworkIsolation } = require('./lib/network-isolation.cjs');
const { renderReport } = require('./lib/render-report.cjs');

function parseArgs(argv) {
  const options = { json: false, help: false, encode: false, out: null };
  for (let i = 0; i < argv.length; i++) {
    if (argv[i] === '--json') options.json = true;
    else if (argv[i] === '--encode') options.encode = true;
    else if (argv[i] === '--help' || argv[i] === '-h') options.help = true;
    else if (argv[i] === '--out') {
      if (!argv[i + 1] || argv[i + 1].startsWith('--')) throw new Error('--out requires a directory.');
      options.out = path.resolve(argv[++i]);
    } else throw new Error(`Unknown argument: ${argv[i]}`);
  }
  return options;
}

function checkProviderOutput(result, provider) {
  if (!result || !Array.isArray(result.cases) || result.cases.length === 0) {
    throw new Error(`${provider} returned no cases.`);
  }
  for (const item of result.cases) {
    if (!/^[a-z0-9][a-z0-9-]*$/.test(item.id || '')) throw new Error(`${provider} returned an invalid case ID.`);
    if (!['native-fixture', 'host-guard'].includes(item.kind)) throw new Error(`Invalid case kind for ${item.id}.`);
    if (!item.label || !item.input || !item.observed) throw new Error(`Incomplete case ${item.id}.`);
    for (const capture of item.captures || []) {
      if (!Number.isSafeInteger(capture.bytes) || capture.bytes < 0 || capture.bytes > 1024 * 1024) {
        throw new Error(`Invalid capture size for ${item.id}.`);
      }
      if (typeof capture.hex !== 'string' || !/^(?:[0-9a-fA-F]{2})*$/.test(capture.hex) || capture.hex.length !== capture.bytes * 2) {
        throw new Error(`Invalid capture bytes for ${item.id}.`);
      }
    }
  }
}

async function run(options) {
  const generatedAt = new Date().toISOString();
  const id = generatedAt.replace(/[:.]/g, '-') + '-' + crypto.randomBytes(3).toString('hex');
  const outputDir = options.out || path.join(__dirname, 'runs', id);
  if (fs.existsSync(outputDir) && (!fs.statSync(outputDir).isDirectory() || fs.readdirSync(outputDir).length)) {
    throw new Error(`Output directory must be new or empty: ${outputDir}`);
  }

  const isolation = installNetworkIsolation();
  try {
    // Loading providers after installing the guard also covers their startup.
    const { runNativeScenarios } = require('./lib/native-fixtures.cjs');
    const { runHostGuardScenarios } = require('./lib/host-guard.cjs');
    const workspaceRoot = path.dirname(__dirname);
    const native = await runNativeScenarios({ workspaceRoot });
    const encoder = options.encode
      ? await require('./lib/encoder-fixtures.cjs').runEncoderScenarios({ sourceCases: native.cases })
      : { cases: [], evidence: [], limitations: [] };
    const host = await runHostGuardScenarios({ workspaceRoot });
    checkProviderOutput(native, 'Native fixtures');
    if (options.encode) checkProviderOutput(encoder, 'Encoder fixtures');
    checkProviderOutput(host, 'Host guard');
    const cases = [...native.cases, ...encoder.cases, ...host.cases];
    if (new Set(cases.map(item => item.id)).size !== cases.length) throw new Error('Duplicate case IDs.');
    if (isolation.attempts.length) throw new Error('A provider attempted a network or process operation; run rejected.');

    const report = {
      schemaVersion: 1,
      name: 'Offline Report Lab',
      generatedAt,
      sourceBuild: '60946132',
      run: {
        id,
        node: process.version,
        kind: 'controlled-offline-fixtures',
        fullGameStarted: false,
        accountQueried: false,
        installedExtensionModified: false,
        encoderExecuted: Boolean(options.encode),
        inputs: 'Locally constructed scenarios, pinned isolated native helpers and pinned page/extension source.'
      },
      networkIsolation: {
        activeDuringRun: true,
        attemptCount: isolation.attempts.length,
        attempts: isolation.attempts,
        transport: options.encode
          ? 'Envelope builders stop at the encoder boundary. Captured inputs are replayed through the isolated encoder with no sender present.'
          : 'Native encoder boundary records locally and returns without sending.',
        scope: 'Process-local API guard, deny-by-default fixture imports, and restricted VM contexts; no OS firewall changes.'
      },
      cases,
      evidence: [...(native.evidence || []), ...(encoder.evidence || []), ...(host.evidence || [])],
      limitations: [...new Set([
        'Controlled fixture inputs are not observations of your gameplay or account.',
        'Only selected helpers execute. This is not the complete client or all reporting paths.',
        'Native envelope captures are before encoding. Optional encoded captures are separately labeled; no server-response equivalence is claimed.',
        'A clean fixture result does not establish multiplayer safety or absence of enforcement.',
        'Historical server reports and private ban rules are unavailable here.',
        ...(native.limitations || []), ...(encoder.limitations || []), ...(host.limitations || [])
      ])]
    };

    fs.mkdirSync(outputDir, { recursive: true });
    for (const item of cases) {
      for (const [index, capture] of (item.captures || []).entries()) {
        const relative = `captures/${item.id}-${index + 1}.bin`;
        fs.mkdirSync(path.join(outputDir, 'captures'), { recursive: true });
        fs.writeFileSync(path.join(outputDir, relative), Buffer.from(capture.hex, 'hex'), { flag: 'wx' });
        capture.file = relative;
      }
    }
    const reportPath = path.join(outputDir, 'report.json');
    const htmlPath = path.join(outputDir, 'index.html');
    fs.writeFileSync(reportPath, JSON.stringify(report, null, 2) + '\n', { flag: 'wx' });
    fs.writeFileSync(htmlPath, renderReport(report), { flag: 'wx' });
    return {
      status: 'complete', outputDir, reportPath, htmlPath,
      caseCount: cases.length,
      captureCount: cases.reduce((count, item) => count + (item.captures || []).length, 0),
      encodedCaptureCount: cases.reduce((count, item) => count + (item.captures || []).filter(capture => capture.stage === 'encoded').length, 0),
      networkAttempts: isolation.attempts.length,
      simulatedInputs: true,
      fullGameStarted: false
    };
  } finally {
    isolation.restore();
  }
}

async function main() {
  const options = parseArgs(process.argv.slice(2));
  if (options.help) {

}

if (require.main === module) main().catch(error => {
  process.stderr.write(`Report Lab: ${error.message}\n`);
  process.exitCode = 1;
});
